<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Book extends Model
{
    protected $fillable = [
        'isbn', 'title', 'slug', 'cover', 'author', 'publisher', 'publication_year',
        'pages', 'description', 'category_id', 'language', 'shelf_location',
    ];

    public function category()
    {
        return $this->belongsTo(Category::class);
    }

    public function copies()
    {
        return $this->hasMany(BookCopy::class);
    }

    public function loans()
    {
        return $this->hasMany(Loan::class);
    }

    public function reviews()
    {
        return $this->hasMany(BookReview::class);
    }

    public function approvedReviews()
    {
        return $this->hasMany(BookReview::class)->whereNotNull('approved_at');
    }

    public function getCoverUrlAttribute(): ?string
    {
        return $this->cover ? asset('storage/' . $this->cover) : null;
    }

    public function getRouteKeyName(): string
    {
        return 'slug';
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\Book;
use App\Models\BookCopy;
use App\Models\Category;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;
use Illuminate\Validation\Rule;

class BookController extends Controller
{
    public function index(Request $request)
    {
        $q = trim((string) $request->query('q'));
        $books = Book::with('category')
            ->withCount(['copies', 'copies as available_count' => fn ($c) => $c->where('status', 'Available')])
            ->when($q !== '', fn ($query) => $query->where(function ($w) use ($q) {
                $w->where('title', 'like', "%$q%")->orWhere('author', 'like', "%$q%")->orWhere('isbn', 'like', "%$q%");
            }))
            ->latest()->paginate(10)->withQueryString();

        return view('admin.books.index', compact('books', 'q'));
    }

    public function create()
    {
        return view('admin.books.create', ['categories' => Category::orderBy('name')->get()]);
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'title'            => 'required|string|max:255',
            'author'           => 'required|string|max:255',
            'publisher'        => 'required|string|max:255',
            'publication_year' => 'required|integer|between:1900,2100',
            'category_id'      => 'required|exists:categories,id',
            'copies'           => 'required|integer|min:1|max:200',
            'isbn'             => 'nullable|string|max:20|unique:books,isbn',
            'pages'            => 'nullable|integer|min:1|max:20000',
            'language'         => 'nullable|string|max:50',
            'shelf_location'   => 'nullable|string|max:50',
            'description'      => 'nullable|string',
            'cover'            => 'nullable|image|mimes:jpeg,png,jpg,webp|max:2048',
        ]);

        $base = Str::slug($data['title']) ?: 'buku';
        $slug = $base;
        $n = 2;
        while (Book::where('slug', $slug)->exists()) {
            $slug = $base . '-' . $n++;
        }

        $coverPath = $request->hasFile('cover') ? $request->file('cover')->store('covers', 'public') : null;

        $book = DB::transaction(function () use ($data, $slug, $coverPath) {
            $book = Book::create([
                'isbn' => $data['isbn'] ?? null,
                'title' => $data['title'],
                'slug' => $slug,
                'cover' => $coverPath,
                'author' => $data['author'],
                'publisher' => $data['publisher'],
                'publication_year' => $data['publication_year'],
                'pages' => $data['pages'] ?? null,
                'description' => $data['description'] ?? null,
                'category_id' => $data['category_id'],
                'language' => $data['language'] ?: 'Indonesia',
                'shelf_location' => $data['shelf_location'] ?? null,
            ]);

            for ($i = 1; $i <= $data['copies']; $i++) {
                BookCopy::create([
                    'book_id' => $book->id,
                    'copy_code' => 'TL-' . str_pad($book->id, 4, '0', STR_PAD_LEFT) . '-' . str_pad($i, 3, '0', STR_PAD_LEFT),
                    'condition' => 'Bagus',
                    'procurement_date' => now()->toDateString(),
                    'status' => 'Available',
                ]);
            }
            return $book;
        });

        return redirect()->route('admin.books.index')
            ->with('success', 'Buku "' . $book->title . '" dengan ' . $data['copies'] . ' eksemplar ditambahkan.');
    }

    public function edit(Book $book)
    {
        $book->loadCount(['copies', 'copies as available_count' => fn ($c) => $c->where('status', 'Available')]);
        return view('admin.books.edit', ['book' => $book, 'categories' => Category::orderBy('name')->get()]);
    }

    public function update(Request $request, Book $book)
    {
        $data = $request->validate([
            'title'            => 'required|string|max:255',
            'author'           => 'required|string|max:255',
            'publisher'        => 'required|string|max:255',
            'publication_year' => 'required|integer|between:1900,2100',
            'category_id'      => 'required|exists:categories,id',
            'add_copies'       => 'nullable|integer|min:0|max:200',
            'isbn'             => ['nullable', 'string', 'max:20', Rule::unique('books', 'isbn')->ignore($book->id)],
            'pages'            => 'nullable|integer|min:1|max:20000',
            'language'         => 'nullable|string|max:50',
            'shelf_location'   => 'nullable|string|max:50',
            'description'      => 'nullable|string',
            'cover'            => 'nullable|image|mimes:jpeg,png,jpg,webp|max:2048',
        ]);

        $coverPath = $book->cover;
        if ($request->hasFile('cover')) {
            if ($book->cover) {
                Storage::disk('public')->delete($book->cover);
            }
            $coverPath = $request->file('cover')->store('covers', 'public');
        } elseif ($request->boolean('remove_cover') && $book->cover) {
            Storage::disk('public')->delete($book->cover);
            $coverPath = null;
        }

        DB::transaction(function () use ($data, $book, $coverPath) {
            // slug tidak diubah supaya tautan lama tetap berlaku
            $book->update([
                'isbn' => $data['isbn'] ?? null,
                'title' => $data['title'],
                'cover' => $coverPath,
                'author' => $data['author'],
                'publisher' => $data['publisher'],
                'publication_year' => $data['publication_year'],
                'pages' => $data['pages'] ?? null,
                'description' => $data['description'] ?? null,
                'category_id' => $data['category_id'],
                'language' => $data['language'] ?: 'Indonesia',
                'shelf_location' => $data['shelf_location'] ?? null,
            ]);

            $add = (int) ($data['add_copies'] ?? 0);
            $n = $book->copies()->count();
            for ($i = 0; $i < $add; $i++) {
                do {
                    $n++;
                    $code = 'TL-' . str_pad($book->id, 4, '0', STR_PAD_LEFT) . '-' . str_pad($n, 3, '0', STR_PAD_LEFT);
                } while (BookCopy::where('copy_code', $code)->exists());

                BookCopy::create([
                    'book_id' => $book->id,
                    'copy_code' => $code,
                    'condition' => 'Bagus',
                    'procurement_date' => now()->toDateString(),
                    'status' => 'Available',
                ]);
            }
        });

        return redirect()->route('admin.books.index')->with('success', 'Buku "' . $book->title . '" diperbarui.');
    }

    public function destroy(Book $book)
    {
        if ($book->cover) {
            Storage::disk('public')->delete($book->cover);
        }
        $book->delete();
        return redirect()->route('admin.books.index')->with('success', 'Buku dihapus.');
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\Category;
use Illuminate\Http\Request;
use Illuminate\Support\Str;
use Illuminate\Validation\Rule;

class CategoryController extends Controller
{
    public function index()
    {
        $categories = Category::withCount('books')->orderBy('name')->get();
        return view('admin.categories.index', compact('categories'));
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'name'        => 'required|string|max:100|unique:categories,name',
            'description' => 'nullable|string|max:255',
        ]);
        $data['slug'] = Str::slug($data['name']);

        if (Category::where('slug', $data['slug'])->exists()) {
            return back()->withErrors(['name' => 'Nama kategori terlalu mirip dengan yang sudah ada.'])->withInput();
        }

        Category::create($data);
        return back()->with('success', 'Kategori "' . $data['name'] . '" ditambahkan.');
    }

    public function edit(Category $category)
    {
        return view('admin.categories.edit', compact('category'));
    }

    public function update(Request $request, Category $category)
    {
        $data = $request->validate([
            'name'        => ['required', 'string', 'max:100', Rule::unique('categories', 'name')->ignore($category->id)],
            'description' => 'nullable|string|max:255',
        ]);
        $slug = Str::slug($data['name']);

        if (Category::where('slug', $slug)->where('id', '!=', $category->id)->exists()) {
            return back()->withErrors(['name' => 'Nama kategori terlalu mirip dengan yang sudah ada.'])->withInput();
        }

        $category->update(['name' => $data['name'], 'description' => $data['description'] ?? null, 'slug' => $slug]);
        return redirect()->route('admin.categories.index')->with('success', 'Kategori diperbarui.');
    }

    public function destroy(Category $category)
    {
        if ($category->books()->exists()) {
            return back()->with('error', 'Kategori masih memiliki buku, tidak bisa dihapus.');
        }
        $category->delete();
        return back()->with('success', 'Kategori dihapus.');
    }
}

<?php

namespace App\Http\Controllers\Student;

use App\Http\Controllers\Controller;
use App\Models\Book;
use App\Models\Category;
use Illuminate\Http\Request;

class BookController extends Controller
{
    private function withStock()
    {
        return Book::with('category')
            ->withCount(['copies', 'copies as available_count' => fn ($c) => $c->where('status', 'Available')]);
    }

    public function index(Request $request)
    {
        $q = trim((string) $request->query('q'));
        $cat = $request->query('kategori');
        $sort = $request->query('urut', 'terbaru');

        $query = $this->withStock()
            ->when($q !== '', fn ($x) => $x->where(function ($w) use ($q) {
                $w->where('title', 'like', "%$q%")->orWhere('author', 'like', "%$q%")->orWhere('publisher', 'like', "%$q%");
            }))
            ->when($cat, fn ($x) => $x->whereHas('category', fn ($c) => $c->where('slug', $cat)));

        $query = match ($sort) {
            'judul' => $query->orderBy('title'),
            'tersedia' => $query->orderByDesc('available_count'),
            default => $query->latest(),
        };

        return view('student.books.index', [
            'books' => $query->paginate(12)->withQueryString(),
            'categories' => Category::withCount('books')->orderBy('name')->get(),
            'q' => $q, 'cat' => $cat, 'sort' => $sort,
        ]);
    }

    public function show(Book $book)
    {
        $book = $this->withStock()->findOrFail($book->id);
        $related = $this->withStock()->where('category_id', $book->category_id)->where('id', '!=', $book->id)->latest()->take(4)->get();
